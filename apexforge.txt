  // ====== PAGE NAVIGATION ======
  function showPage(pageId, btn) {
    document.querySelectorAll('.page-section').forEach(p => p.classList.remove('active'));
    const target = document.getElementById('page-' + pageId);
    if (target) target.classList.add('active');
    document.querySelectorAll('.page-nav-btn').forEach(b => b.classList.remove('active'));
    if (btn) {
      btn.classList.add('active');
    } else {
      // Find matching btn by onclick attribute
      document.querySelectorAll('.page-nav-btn').forEach(b => {
        if (b.getAttribute('onclick') && b.getAttribute('onclick').includes("'" + pageId + "'")) b.classList.add('active');
      });
    }
    document.getElementById('pageNavWrap').scrollIntoView({ behavior: 'smooth', block: 'start' });
    target && target.querySelectorAll('.fade-up').forEach((el, i) => {
      el.classList.remove('visible');
      setTimeout(() => el.classList.add('visible'), i * 80 + 100);
    });

  function calcMacros() {
    const w = parseFloat(document.getElementById('macroWeight').value);
    if (!w) {
      ['macroCalories','macroProtein','macroCarbs','macroFat'].forEach(id => document.getElementById(id).textContent = '—');
      document.getElementById('macroNote').style.display = 'none';
      return;
    }
    const goal = document.getElementById('macroGoal').value;
    const activity = parseFloat(document.getElementById('macroActivity').value);
    const bmr = w * 22;
    let tdee = bmr * activity;
    let adj = goal === 'cut' ? -500 : goal === 'bulk' ? 300 : 0;
    const kcal = Math.round(tdee + adj);
    const protein = Math.round(goal === 'cut' ? w * 2.4 : goal === 'bulk' ? w * 2.2 : w * 2.6);
    const fat = Math.round(goal === 'cut' ? w * 0.8 : goal === 'bulk' ? w * 1.0 : w * 0.9);
    const carbCal = kcal - (protein * 4) - (fat * 9);
    const carbs = Math.max(0, Math.round(carbCal / 4));
    document.getElementById('macroCalories').textContent = kcal;
    document.getElementById('macroProtein').textContent = protein;
    document.getElementById('macroCarbs').textContent = carbs;
    document.getElementById('macroFat').textContent = fat;
    const notes = {
      cut: '🔥 Cut phase: Prioritise protein to preserve muscle. Eat most carbs around training. Track weekly — adjust if loss stalls for 2+ weeks.',
      maintain: '⚖️ Maintenance: Focus on food quality and consistency. This is your baseline — perfect for body recomposition when paired with progressive training.',
      bulk: '💪 Lean bulk: Aim for 0.25–0.5 kg/week weight gain. If gaining faster, reduce carbs by 50g. Re-assess macros every 4 weeks as weight changes.'
    };
    const noteEl = document.getElementById('macroNote');
    noteEl.textContent = notes[goal];
    noteEl.style.display = 'block';
  }

  // ====== HEIGHT POTENTIAL CALCULATOR ======
  function calcHeightPotential() {
    const current = parseFloat(document.getElementById('hcCurrentHeight').value);
    const father = parseFloat(document.getElementById('hcFatherHeight').value);
    const mother = parseFloat(document.getElementById('hcMotherHeight').value);
    const age    = parseFloat(document.getElementById('hcAge').value);
    const gender = document.getElementById('hcGender').value;
    const resultsBox = document.getElementById('hcResults');

    if (!current || !father || !mother || !age) {
      resultsBox.style.opacity = '0.3';
      return;
    }
    resultsBox.style.opacity = '1';

    // Mid-parental height (genetic ceiling)
    let genetic = gender === 'male'
      ? ((father + mother + 13) / 2)
      : ((father + mother - 13) / 2);
    genetic = Math.round(genetic * 10) / 10;

    // Programme potential: posture + decompression gains
    // Age factor: under 20 more gain possible, over 35 less
    let gainFactor = age < 18 ? 5.0 : age < 22 ? 4.0 : age < 28 ? 3.0 : age < 35 ? 2.5 : 1.5;
    // If already near genetic ceiling, reduce gain
    const gap = Math.max(0, genetic - current);
    const postureGain = Math.min(gap * 0.6 + gainFactor * 0.4, gainFactor);
    const potential = Math.round((current + postureGain) * 10) / 10;

    document.getElementById('hcGenetic').textContent = genetic;
    document.getElementById('hcPotential').textContent = potential;

    // Bars — normalise to max possible
    const maxH = Math.max(potential, genetic, current) + 5;
    const minH = Math.min(current) - 5;
    const range = maxH - minH;

    const pct = v => Math.round(((v - minH) / range) * 100);
    document.getElementById('hcBarCurrent').style.width = pct(current) + '%';
    document.getElementById('hcBarGenetic').style.width = pct(genetic) + '%';
    document.getElementById('hcBarPotential').style.width = pct(potential) + '%';
    document.getElementById('hcBarCurrentVal').textContent = current;
    document.getElementById('hcBarGeneticVal').textContent = genetic;
    document.getElementById('hcBarPotentialVal').textContent = potential;

    const gained = Math.round((potential - current) * 10) / 10;
    const advice = document.getElementById('hcAdvice');
    advice.style.display = 'block';
    if (age < 20) {
      advice.innerHTML = `🌱 <strong style="color:var(--text);">Growth Window Active.</strong> At ${age} years old you still have active growth plates. Our Growth Axis Training can maximise your remaining natural growth AND correct posture — potentially adding up to <strong style="color:var(--orange);">+${gained}cm</strong> to your current height.`;
    } else if (gained >= 3) {
      advice.innerHTML = `📏 <strong style="color:var(--text);">High Potential Detected.</strong> Our spinal decompression and posture correction protocol estimates a gain of up to <strong style="color:var(--orange);">+${gained}cm</strong> — primarily through vertebral spacing and eliminating postural height loss. Most clients see visible results within 6–8 weeks.`;
    } else {
      advice.innerHTML = `📐 <strong style="color:var(--text);">Achievable Gains.</strong> An estimated <strong style="color:var(--orange);">+${gained}cm</strong> is achievable through daily spinal decompression, posture realignment, and targeted flexibility work. Even 1–2cm can dramatically change how tall you appear and how you carry yourself.`;
    }
  }

  function clearPlan() {
    const output = document.getElementById('planOutput');
    output.style.display = 'none';
    output.innerHTML = '';
    selected.clear();

  // ====== TESTIMONIALS SLIDER (fixed for hidden sections) ======
  let testimonialIndex = 0;
  function slideTestimonials(dir) {
    const track = document.getElementById('testimonialsTrack');
    const cards = track.querySelectorAll('.testimonial-card');
    const max = Math.max(0, cards.length - 1);
    testimonialIndex = Math.max(0, Math.min(testimonialIndex + dir, max));
    // Use actual measured width when visible
    const cardEl = cards[0];
    const w = cardEl.getBoundingClientRect().width || parseInt(getComputedStyle(cardEl).minWidth) || 380;
    track.style.transform = `translateX(-${testimonialIndex * (w + 30)}px)`;
    // Update dots
    document.querySelectorAll('.t-dot').forEach((d, i) => d.classList.toggle('active', i === testimonialIndex));
  }

  // Add dots to testimonials when results page is opened
  function initTestimonialDots() {
    const controls = document.querySelector('.testimonial-controls');
    if (!controls || controls.querySelector('.t-dot')) return;
    const track = document.getElementById('testimonialsTrack');
    const count = track.querySelectorAll('.testimonial-card').length;
    const dotsWrap = document.createElement('div');
    dotsWrap.style.cssText = 'display:flex;gap:8px;align-items:center;';
    for (let i = 0; i < count; i++) {
      const dot = document.createElement('button');
      dot.className = 't-dot';
      dot.style.cssText = `width:8px;height:8px;border-radius:50%;border:none;cursor:pointer;transition:all 0.3s;background:${i===0?'var(--red)':'var(--light-gray)'};padding:0;`;
      dot.onclick = () => { testimonialIndex = i - 1; slideTestimonials(1); };
      dotsWrap.appendChild(dot);
    }
    controls.appendChild(dotsWrap);
    // Add dot active style
    const style = document.createElement('style');
    style.textContent = '.t-dot.active{background:var(--red) !important;transform:scale(1.3);}';
    document.head.appendChild(style);
    document.querySelector('.t-dot')?.classList.add('active');
  }

  // Patch showPage to init testimonial dots when results page opens
  const _origShowPage = showPage;
  window.showPage = function(pageId, btn) {
    _origShowPage(pageId, btn);
    if (pageId === 'results') setTimeout(initTestimonialDots, 150);
    if (pageId === 'nutrition') calcMacros();
  };

  // ====== CONTACT FORM ======
  function handleSubmit() {
    const inputs = document.querySelectorAll('#contact input, #contact textarea');
    let allFilled = true;
    inputs.forEach(inp => {
      if (!inp.value.trim()) {
        allFilled = false;
        inp.style.borderBottomColor = '#7BBFF7';
        setTimeout(() => inp.style.borderBottomColor = '', 2000);
      }
    });
    if (!allFilled) return;
    const btn = document.querySelector('#page-contact .btn-primary');
    const original = btn.textContent;
    btn.textContent = 'Sending...';
    btn.style.opacity = '0.7';
    btn.disabled = true;
    setTimeout(() => {
      inputs.forEach(inp => inp.value = '');
      btn.textContent = original;
      btn.style.opacity = '';
      btn.disabled = false;
      const success = document.getElementById('formSuccess');
      success.style.display = 'block';
      setTimeout(() => success.style.display = 'none', 4000);
    }, 1200);
  }

  (function () {
    const navUl = document.querySelector('nav ul');
    const burger = document.createElement('button');
    burger.innerHTML = '☰';
    burger.style.cssText = 'display:none;background:none;border:none;color:#FFF3B0;font-size:24px;cursor:pointer;padding:0;';

      if (window.innerWidth <= 900) {
        burger.style.display = 'block';
        navUl.style.cssText = 'display:none;flex-direction:column;position:absolute;top:70px;left:0;right:0;background:rgba(5,5,5,0.98);padding:20px 30px;gap:20px;z-index:999;';
        burger.onclick = () => { const vis = navUl.style.display === 'flex'; navUl.style.display = vis ? 'none' : 'flex'; burger.innerHTML = vis ? '☰' : '✕'; };
      } else {
        burger.style.display = 'none';
        navUl.style.cssText = '';
      }
    }
    window.addEventListener('resize', checkMobile);
    checkMobile();
    navUl.querySelectorAll('a').forEach(a => a.addEventListener('click', () => { if (window.innerWidth <= 900) { navUl.style.display = 'none'; burger.innerHTML = '☰'; } }));
  })();
